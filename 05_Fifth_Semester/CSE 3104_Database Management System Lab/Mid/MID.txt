# 🔹 1. BASIC SELECT

```sql
-- ALL DEVELOPERS
SELECT * FROM DEVELOPERS;

-- SELECT SPECIFIC COLUMNS
SELECT DEVNAME, DEPARTMENT FROM DEVELOPERS;

-- DISTINCT DEPARTMENTS
SELECT DISTINCT DEPARTMENT FROM DEVELOPERS;
```

---

# 🔹 2. WHERE CONDITIONS

```sql
-- DEVELOPERS WITH SALARY > 60000
SELECT * FROM DEVELOPERS
WHERE BASESALARY > 60000;

-- BACKEND DEVELOPERS
SELECT * FROM DEVELOPERS
WHERE DEPARTMENT = 'BACKEND';

-- PROJECTS WITH BUDGET BETWEEN 300K AND 700K
SELECT * FROM PROJECT
WHERE BUDGET BETWEEN 300000 AND 700000;

-- ACTIVE PROJECTS
SELECT * FROM PROJECT
WHERE STATUS = 'ACTIVE';
```

---

# 🔹 3. ORDER BY

```sql
-- SORT BY SALARY ASC
SELECT * FROM DEVELOPERS
ORDER BY BASESALARY ASC;

-- SORT BY SALARY DESC
SELECT * FROM DEVELOPERS
ORDER BY BASESALARY DESC;
```

---

# 🔹 4. AGGREGATE FUNCTIONS

```sql
-- TOTAL DEVELOPERS
SELECT COUNT(*) FROM DEVELOPERS;

-- AVG SALARY
SELECT AVG(BASESALARY) FROM DEVELOPERS;

-- MAX SALARY
SELECT MAX(BASESALARY) FROM DEVELOPERS;

-- MIN SALARY
SELECT MIN(BASESALARY) FROM DEVELOPERS;

-- TOTAL BUDGET
SELECT SUM(BUDGET) FROM PROJECT;
```

---

# 🔹 5. GROUP BY

```sql
-- COUNT DEVELOPERS BY DEPARTMENT
SELECT DEPARTMENT, COUNT(*) 
FROM DEVELOPERS
GROUP BY DEPARTMENT;

-- AVG SALARY BY DEPARTMENT
SELECT DEPARTMENT, AVG(BASESALARY)
FROM DEVELOPERS
GROUP BY DEPARTMENT;

-- PROJECT COUNT BY STATUS
SELECT STATUS, COUNT(*)
FROM PROJECT
GROUP BY STATUS;
```

---

# 🔹 6. HAVING

```sql
-- DEPARTMENTS WITH AVG SALARY > 60000
SELECT DEPARTMENT, AVG(BASESALARY)
FROM DEVELOPERS
GROUP BY DEPARTMENT
HAVING AVG(BASESALARY) > 60000;
```

---

# 🔹 7. JOINS (VERY IMPORTANT)

```sql
-- INNER JOIN
SELECT D.DEVNAME, P.PROJECTNAME
FROM PROJECTALLOCATIONS PA
JOIN DEVELOPERS D ON PA.DEVID = D.DEVID
JOIN PROJECT P ON PA.PROJECTID = P.PROJECTID;

-- LEFT JOIN (ALL DEVELOPERS)
SELECT D.DEVNAME, P.PROJECTNAME
FROM DEVELOPERS D
LEFT JOIN PROJECTALLOCATIONS PA ON D.DEVID = PA.DEVID
LEFT JOIN PROJECT P ON PA.PROJECTID = P.PROJECTID;

-- RIGHT JOIN
SELECT D.DEVNAME, P.PROJECTNAME
FROM PROJECTALLOCATIONS PA
RIGHT JOIN PROJECT P ON PA.PROJECTID = P.PROJECTID
LEFT JOIN DEVELOPERS D ON PA.DEVID = D.DEVID;
```

---

# 🔹 8. MULTI-TABLE QUERY

```sql
-- FULL DETAILS
SELECT 
    D.DEVNAME,
    D.DEPARTMENT,
    P.PROJECTNAME,
    P.CLIENT,
    PA.ALLOCHOURS,
    PA.ROLE
FROM PROJECTALLOCATIONS PA
JOIN DEVELOPERS D ON PA.DEVID = D.DEVID
JOIN PROJECT P ON PA.PROJECTID = P.PROJECTID;
```

---

# 🔹 9. SUBQUERIES

```sql
-- DEVELOPERS WITH MAX SALARY
SELECT * FROM DEVELOPERS
WHERE BASESALARY = (SELECT MAX(BASESALARY) FROM DEVELOPERS);

-- PROJECT WITH HIGHEST BUDGET
SELECT * FROM PROJECT
WHERE BUDGET = (SELECT MAX(BUDGET) FROM PROJECT);

-- DEVELOPERS WHO WORK IN PROJECT P1
SELECT * FROM DEVELOPERS
WHERE DEVID IN (
    SELECT DEVID FROM PROJECTALLOCATIONS WHERE PROJECTID = 'P1'
);
```

---

# 🔹 10. EXISTS / NOT EXISTS

```sql
-- DEVELOPERS WHO ARE ALLOCATED
SELECT * FROM DEVELOPERS D
WHERE EXISTS (
    SELECT 1 FROM PROJECTALLOCATIONS PA
    WHERE PA.DEVID = D.DEVID
);

-- DEVELOPERS WITH NO PROJECT
SELECT * FROM DEVELOPERS D
WHERE NOT EXISTS (
    SELECT 1 FROM PROJECTALLOCATIONS PA
    WHERE PA.DEVID = D.DEVID
);
```

---

# 🔹 11. LIKE (SEARCH)

```sql
-- NAMES START WITH 'A'
SELECT * FROM DEVELOPERS
WHERE DEVNAME LIKE 'A%';

-- EMAIL FROM TECH.COM
SELECT * FROM DEVELOPERS
WHERE EMAIL LIKE '%TECH.COM';
```

---

# 🔹 12. CASE (CONDITIONAL)

```sql
SELECT DEVNAME,
CASE
    WHEN BASESALARY > 80000 THEN 'HIGH'
    WHEN BASESALARY BETWEEN 50000 AND 80000 THEN 'MEDIUM'
    ELSE 'LOW'
END AS SALARY_LEVEL
FROM DEVELOPERS;
```

---

# 🔹 13. UPDATE / DELETE

```sql
-- UPDATE SALARY
UPDATE DEVELOPERS
SET BASESALARY = BASESALARY + 5000
WHERE DEPARTMENT = 'FRONTEND';

-- DELETE A PROJECT
DELETE FROM PROJECT
WHERE PROJECTID = 'P4';
```

---

# 🔹 14. VIEW CREATION

```sql
CREATE VIEW DEV_PROJECT_VIEW AS
SELECT D.DEVNAME, P.PROJECTNAME, PA.ROLE
FROM PROJECTALLOCATIONS PA
JOIN DEVELOPERS D ON PA.DEVID = D.DEVID
JOIN PROJECT P ON PA.PROJECTID = P.PROJECTID;
```

---

# 🔹 15. TOP / LIMIT

```sql
-- TOP 3 HIGHEST SALARY
SELECT TOP 3 * FROM DEVELOPERS
ORDER BY BASESALARY DESC;
```

---

# 🔥 BONUS (EXAM-LEVEL HARD)

```sql
-- DEVELOPER WITH MOST WORK HOURS
SELECT TOP 1 D.DEVNAME, SUM(PA.ALLOCHOURS) AS TOTAL_HOURS
FROM PROJECTALLOCATIONS PA
JOIN DEVELOPERS D ON PA.DEVID = D.DEVID
GROUP BY D.DEVNAME
ORDER BY TOTAL_HOURS DESC;

-- PROJECT WITH HIGHEST TOTAL HOURS
SELECT TOP 1 P.PROJECTNAME, SUM(PA.ALLOCHOURS) AS TOTAL
FROM PROJECTALLOCATIONS PA
JOIN PROJECT P ON PA.PROJECTID = P.PROJECTID
GROUP BY P.PROJECTNAME
ORDER BY TOTAL DESC;
```

# 🔷 1. BASIC JOIN VARIATIONS

```sql
-- ALL DEVELOPERS WITH THEIR PROJECTS
```



# 🔷 2. FILTER WITH JOIN

```sql
-- BACKEND DEVELOPERS WORKING ON ACTIVE PROJECTS

FROM DEVELOPERS D

JOIN PROJECT P ON PA.PROJECTID = P.PROJECTID
```

